// Comando para gerar certificado SSL autoassinado para PostgreSQL no Windows
"C:\Program Files\OpenSSL-Win64\bin\openssl.exe" req -new -x509 -days 365 -nodes -text -out "C:\Program Files\PostgreSQL\18\data\server.crt" -keyout "C:\Program Files\PostgreSQL\18\data\server.key"
// o comando acima gera um certificado SSL autoassinado válido por 365 dias e salva os arquivos server.crt e server.key no diretório de dados do PostgreSQL.
